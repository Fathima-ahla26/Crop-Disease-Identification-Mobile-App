import { useRouter } from "expo-router";
import { useState } from "react";
import {
  Modal,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,
} from "react-native";

export default function LoginScreen() {
  // Secure database-based login is available at /secure-auth.

  const router = useRouter();
  const [isSignUp, setIsSignUp] = useState(false);

  // Default farmer account (Default නම Farmer Kamal)
  const [registeredFarmer, setRegisteredFarmer] = useState({
    name: "Kamal",
    emailOrPhone: "farmer@cropcare.com",
    password: "crop123",
  });

  // Form states
  const [name, setName] = useState("");
  const [emailOrPhone, setEmailOrPhone] = useState("");
  const [password, setPassword] = useState("");
  const [errorMessage, setErrorMessage] = useState("");
  const [successMessage, setSuccessMessage] = useState("");

  // Forgot password modal states
  const [forgotModalVisible, setForgotModalVisible] = useState(false);
  const [resetEmail, setResetEmail] = useState("");
  const [resetStatus, setResetStatus] = useState("");

  const handleAuth = () => {
    setErrorMessage("");
    setSuccessMessage("");

    if (isSignUp) {
      if (!name.trim()) {
        setErrorMessage("Please enter your full name.");
        return;
      }
      if (!emailOrPhone.trim()) {
        setErrorMessage("Please enter your email or phone number.");
        return;
      }
      if (!password || password.length < 4) {
        setErrorMessage("Password must be at least 4 characters long.");
        return;
      }

      // අලුත් ගොවියාගේ විස්තර සහ නම save කරගැනීම
      setRegisteredFarmer({
        name: name.trim(),
        emailOrPhone: emailOrPhone.trim(),
        password: password,
      });

      setSuccessMessage(
        "Farmer account registered successfully! Please log in.",
      );
      setIsSignUp(false);
      setPassword("");
      return;
    }

    if (!emailOrPhone.trim()) {
      setErrorMessage("Please enter your email or phone number.");
      return;
    }
    if (!password) {
      setErrorMessage("Please enter your password.");
      return;
    }

    // Credentials පරීක්ෂාව
    const isIdentifierCorrect =
      emailOrPhone.trim().toLowerCase() ===
      registeredFarmer.emailOrPhone.toLowerCase();
    const isPasswordCorrect = password === registeredFarmer.password;

    if (!isIdentifierCorrect || !isPasswordCorrect) {
      setErrorMessage(
        "Invalid credentials. Please enter your correct registered details.",
      );
      return;
    }

    // ලියාපදිංචි වූ නම Home screen එකට යැවීම
    router.replace({
      pathname: "/home",
      params: { farmerName: registeredFarmer.name },
    } as any);
  };

  const handlePasswordReset = () => {
    if (!resetEmail.trim()) {
      setResetStatus("Please enter your registered email address.");
      return;
    }
    setResetStatus("Password reset link has been sent to your email.");
    setTimeout(() => {
      setForgotModalVisible(false);
      setResetEmail("");
      setResetStatus("");
      setSuccessMessage(
        "Password reset instructions sent. Please check your inbox.",
      );
    }, 1800);
  };

  return (
    <View style={styles.container}>
      <View style={styles.topCurveBackground}>
        <View style={styles.decorCircleLarge} />
        <View style={styles.decorCircleSmall} />
      </View>

      <ScrollView
        contentContainerStyle={styles.scrollContent}
        showsVerticalScrollIndicator={false}
      >
        <View style={styles.brandHeader}>
          <View style={styles.logoCircle}>
            <Text style={styles.logoEmoji}>🌱</Text>
          </View>
          <Text style={styles.brandName}>CropCare</Text>
        </View>

        <View style={styles.tabContainer}>
          <TouchableOpacity
            style={[styles.tabBtn, !isSignUp && styles.tabActive]}
            onPress={() => {
              setIsSignUp(false);
              setErrorMessage("");
              setSuccessMessage("");
            }}
          >
            <Text style={[styles.tabText, !isSignUp && styles.tabTextActive]}>
              Login
            </Text>
          </TouchableOpacity>

          <TouchableOpacity
            style={[styles.tabBtn, isSignUp && styles.tabActive]}
            onPress={() => {
              setIsSignUp(true);
              setErrorMessage("");
              setSuccessMessage("");
            }}
          >
            <Text style={[styles.tabText, isSignUp && styles.tabTextActive]}>
              Sign Up
            </Text>
          </TouchableOpacity>
        </View>

        <Text style={styles.title}>
          {isSignUp ? "Create Farmer Account" : "Farmer Login"}
        </Text>
        <Text style={styles.subtitle}>
          {isSignUp
            ? "Sign up to track crops, analyze leaves, and consult officers."
            : "Enter your registered details to access your farm advisory."}
        </Text>

        {successMessage ? (
          <View style={styles.successBox}>
            <Text style={styles.successText}>✅ {successMessage}</Text>
          </View>
        ) : null}

        {errorMessage ? (
          <View style={styles.errorBox}>
            <Text style={styles.errorText}>⚠️ {errorMessage}</Text>
          </View>
        ) : null}

        <View style={styles.form}>
          {isSignUp && (
            <View style={styles.inputContainer}>
              <Text style={styles.inputIcon}>👤</Text>
              <TextInput
                style={styles.input}
                placeholder="Full Name (e.g. Kamal Perera)"
                placeholderTextColor="#9CA3AF"
                value={name}
                onChangeText={setName}
              />
            </View>
          )}

          <View style={styles.inputContainer}>
            <Text style={styles.inputIcon}>✉️</Text>
            <TextInput
              style={styles.input}
              placeholder="Email or Phone Number"
              placeholderTextColor="#9CA3AF"
              value={emailOrPhone}
              onChangeText={setEmailOrPhone}
              autoCapitalize="none"
            />
          </View>

          <View style={styles.inputContainer}>
            <Text style={styles.inputIcon}>🔒</Text>
            <TextInput
              style={styles.input}
              placeholder="Password"
              placeholderTextColor="#9CA3AF"
              secureTextEntry
              value={password}
              onChangeText={setPassword}
            />
          </View>

          {!isSignUp && (
            <TouchableOpacity
              style={styles.forgotBtn}
              onPress={() => {
                setForgotModalVisible(true);
                setResetStatus("");
              }}
            >
              <Text style={styles.forgotText}>Forgot Password?</Text>
            </TouchableOpacity>
          )}

          <TouchableOpacity
            style={styles.authBtn}
            onPress={handleAuth}
            activeOpacity={0.88}
          >
            <Text style={styles.authBtnText}>
              {isSignUp ? "Register Account" : "Login"}
            </Text>
          </TouchableOpacity>

          <Text style={styles.orText}>or continue with</Text>

          <TouchableOpacity
            style={styles.googleBtn}
            onPress={() =>
              router.replace({
                pathname: "/home",
                params: { farmerName: "Kamal" },
              } as any)
            }
            activeOpacity={0.85}
          >
            <Text style={styles.googleIcon}>G</Text>
            <Text style={styles.googleBtnText}>Continue with Google</Text>
          </TouchableOpacity>
        </View>
      </ScrollView>

      {/* Forgot Password Modal */}
      <Modal visible={forgotModalVisible} transparent animationType="fade">
        <View style={styles.modalOverlay}>
          <View style={styles.modalContent}>
            <Text style={styles.modalTitle}>Reset Password</Text>
            <Text style={styles.modalSubtitle}>
              Enter your registered email address to receive password reset
              instructions.
            </Text>

            {resetStatus ? (
              <Text
                style={[
                  styles.modalStatusText,
                  resetStatus.includes("sent")
                    ? styles.textSuccess
                    : styles.textDanger,
                ]}
              >
                {resetStatus}
              </Text>
            ) : null}

            <View style={styles.inputContainer}>
              <Text style={styles.inputIcon}>✉️</Text>
              <TextInput
                style={styles.input}
                placeholder="Registered Email"
                placeholderTextColor="#9CA3AF"
                value={resetEmail}
                onChangeText={setResetEmail}
                autoCapitalize="none"
              />
            </View>

            <View style={styles.modalActions}>
              <TouchableOpacity
                style={styles.modalCancelBtn}
                onPress={() => setForgotModalVisible(false)}
              >
                <Text style={styles.modalCancelText}>Cancel</Text>
              </TouchableOpacity>

              <TouchableOpacity
                style={styles.modalSubmitBtn}
                onPress={handlePasswordReset}
              >
                <Text style={styles.modalSubmitText}>Send Link</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </View>
  );
}
